Zapytanie 1. SELECT marka, model, cena FROM pojazdy WHERE marka = "BM" ORDER BY cena DESC LIMIT 15;
Zapytanie 2. SELECT AVG(cena) AS "Średnia cena", MAX(cena) AS "Maksymalna cena" FROM pojazdy WHERE model = "meta";
Zapytanie 3. SELECT marka, model, cena, nazwa, doplata FROM pojazdy INNER JOIN kolory ON kolor = kolory.id WHERE model = "alfa";
Zapytanie 4. SELECT marka, model, cena FROM pojazdy ORDER BY RAND() LIMIT 2;
